## Діаграма процесу реєстрації (Mermaid)

```mermaid
sequenceDiagram
    actor User as Користувач
    participant UI as Веб-інтерфейс
    participant API as Backend API
    participant DB as База даних
    participant Email as Email Service
    participant Cache as Redis Cache

    User->>UI: Відкрити форму реєстрації
    UI->>User: Відобразити форму
    
    User->>UI: Ввести дані (назва, email, пароль)
    UI->>UI: Валідація на клієнті
    
    User->>UI: Натиснути "Зареєструватися"
    UI->>API: POST /api/auth/register
    
    API->>Cache: Перевірити rate limit
    Cache-->>API: OK
    
    API->>DB: Перевірити унікальність email
    DB-->>API: Email не існує
    
    API->>DB: Перевірити унікальність назви
    DB-->>API: Назва не існує
    
    API->>API: Хешувати пароль (bcrypt)
    API->>DB: Створити організацію
    DB-->>API: Організація створена
    
    API->>DB: Створити адміністратора
    DB-->>API: Користувач створений
    
    API->>DB: Створити токен підтвердження
    DB-->>API: Токен створено
    
    API->>Email: Відправити лист підтвердження
    Email-->>API: Email відправлено
    
    API->>Cache: Зберегти сесію
    Cache-->>API: OK
    
    API-->>UI: 201 Created (success)
    UI-->>User: Показати повідомлення про успіх
    
    Note over User,Email: Email підтвердження
    Email->>User: Лист з посиланням
    User->>UI: Клік на посилання
    UI->>API: GET /api/auth/verify?token=xxx
    API->>DB: Підтвердити email
    DB-->>API: OK
    API-->>UI: Redirect to login
    UI-->>User: Організація активована!

// Приклад middleware для rate limiting
func RateLimitMiddleware(limit int, window time.Duration) gin.HandlerFunc {
    limiter := rate.NewLimiter(rate.Every(window), limit)
    
    return func(c *gin.Context) {
        ip := c.ClientIP()
        
        if !limiter.Allow() {
            c.JSON(http.StatusTooManyRequests, gin.H{
                "error": "Too many registration attempts. Please try again later.",
                "retry_after": window.Seconds(),
            })
            c.Abort()
            return
        }
        
        c.Next()
    }
}

// Приклад асинхронної обробки реєстрації
func (s *AuthService) RegisterOrganizationAsync(ctx context.Context, req RegisterRequest) error {
    // 1. Швидка валідація та відповідь користувачу
    if err := s.validateRegistration(req); err != nil {
        return err
    }
    
    // 2. Додати в чергу для обробки
    job := &RegistrationJob{
        OrganizationName: req.Name,
        Email: req.Email,
        PasswordHash: s.hashPassword(req.Password),
        Timestamp: time.Now(),
    }
    
    return s.queue.Publish("registration.queue", job)
}
```
